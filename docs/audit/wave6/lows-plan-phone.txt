## The script
Write <scratchpad>\lows\plan_fix_a_false.py, in exactly this pattern (copy it):

```python
import sys
import openpyxl
sys.stdout.reconfigure(encoding="utf-8")
PLAN = sys.argv[1] if len(sys.argv) > 1 else r"D:\Projects\androidVersion\qa\NoteOrganizer_Android_TestPlan.xlsx"
TITLE, PRE, STEPS, EXPECTED, AUTO = 4, 6, 7, 8, 15
FRAGMENTS = [
    ("6", AUTO, "old text, exactly as in the cell", "new text"),
]
def main():
    wb = openpyxl.load_workbook(PLAN)
    ws = wb["Test Cases"]
    rows = {str(r[0].value): r for r in ws.iter_rows(min_row=2)}
    for case, col, old, new in FRAGMENTS:
        cell = rows[case][col - 1]
        text = str(cell.value)
        if text.count(old) != 1:
            raise SystemExit("case %s col %d: %r found %d times" % (case, col, old[:60], text.count(old)))
        cell.value = text.replace(old, new)
    wb.save(PLAN)
    print("fragments", len(FRAGMENTS))
main()
```
